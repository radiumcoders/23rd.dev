## Usage

Pass the scroll container if the page scrolls inside an element, not the window.

<FrameworkCode>
  <FrameworkReact>
```tsx
"use client"

import { useRef } from "react"
import { RadiantLines } from "@/components/ui/radiant-lines"

export function Hero() {
  const ref = useRef<HTMLDivElement>(null)

  return (
    <div ref={ref} className="relative h-svh overflow-y-auto">
      <RadiantLines containerRef={ref} className="sticky top-0 h-svh" />
      <div className="relative z-10 p-10">Your content</div>
    </div>
  )
}
```
  </FrameworkReact>
  <FrameworkSvelte>
```svelte
<script>
  import RadiantLines from "$lib/components/ui/radiant-lines.svelte"

  let scroller: HTMLDivElement | undefined = $state()
</script>

<div bind:this={scroller} class="relative h-svh overflow-y-auto">
  <RadiantLines container={scroller} class="sticky top-0 h-svh" />
  <div class="relative z-10 p-10">Your content</div>
</div>
```
  </FrameworkSvelte>
</FrameworkCode>

## Props

| Prop | Type | Default |
| --- | --- | --- |
| `colors` | `string[]` | coral / teal / amber / blue / pink / lime / slate |
| `starCount` | `number` | `420` |
| `displacement` | `number` | `1`, star jump on warp |
| `containerRef` | `RefObject<HTMLElement \| null>` | React: window scroll |
| `container` | `HTMLElement` | Svelte: window scroll |
| `className` | `string` | React only |
| `class` | `string` | Svelte only |
